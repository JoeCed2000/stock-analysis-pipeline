FRONTEND (React + Vite — port 5180)
══════════════════════════════════════════════════════════

cd /mnt/c/Users/cedon/Documents/Codex/stock-analysis-pipeline/frontend

# Install dépendances (première fois)
npm install

# Dev server
npm run dev -- --host 0.0.0.0 --port 5180

# Build production
npm run build

# Preview build

cd /mnt/c/Users/cedon/Documents/Codex/stock-analysis-pipeline/backend && PYTHONPATH=.. uvicorn main:app --host 0.0.0.0 --port 8780 --reload &
